# Fireworks Reimagined Settings

# Maximum number of rockets allowed above a firework spawner.
# Use this to prevent particle spam from too many rockets in one area.
fireworks_max_rockets (Max number of rockets above a firework spawner) int 2 0 100

# Radius used when counting nearby rockets for fireworks_max_rockets.
fireworks_scan_radius (Firework search radius) int 12 1 64

# When disabled, fireworks cannot be crafted, but existing fireworks can still be dyed.
fireworks_enable_crafting (Enable crafting recipes for fireworks) bool true

# Comma-separated list of standard firework shapes to register.
# Available: sphere, star, ring, burst, spiral, chaotic, flame, snowflake, hour_glass
# Leave empty to enable all shapes. Disabled shapes are unregistered.
fireworks_enabled_shapes (Enabled firework shapes) string sphere,star,ring,burst,spiral,chaotic,flame,snowflake,hour_glass

# When enabled, item and node names for shapes removed from
# fireworks_enabled_shapes are aliased to the first enabled shape (preferring
# burst). When disabled, removed shapes are unregistered with no alias.
fireworks_alias_disabled_shapes (Alias disabled firework shapes) bool false

# When enabled, fireworks respond to mesecons and MCL redstone input and emit output pulses.
fireworks_enable_mesecons (Enable redstone/mesecons support) bool true

# When enabled, every 10 mesecon/redstone activations require a manual right-click reload.
fireworks_manual_refill (Require manual right-click reload after 10 activations) bool true
